% ── 1. Capçalera del full: buida per defecte ─────────────────────────
% El projecte no porta cap dada de cap centre. Si a la carpeta hi ha un
% capsalera.tex que la redefineixi, main.tex l'incorpora tot sol.
\providecommand{\capsaleraexamen}{}

% ── 2. Ela geminada (l·l) ────────────────────────────────────────────
\DeclareUnicodeCharacter{00B7}{\kern-0.1em\textperiodcentered\kern-0.1em}

% ── 3. Fórmules: les fraccions dels exercicis respiren ───────────────
% Una mica més d'aire entre línies, perquè les \frac en \displaystyle no
% toquin la línia de sobre.
\setlength{\jot}{6pt}
\renewcommand{\arraystretch}{1.3}

% ── 4. Enunciat opcional ─────────────────────────────────────────────
% \enunciat{Calcula:} abans de \input{exN.tex}, al main.tex.
\newcommand{\enunciat}[1]{\noindent\textbf{#1}\par\medskip}
